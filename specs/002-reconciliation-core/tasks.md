```ts
// packages/domain/src/reconciliations.ts
export const RECONCILIATION_DATE_WINDOW_DAYS = 2;
export const RECONCILIATION_CENTS_TOLERANCE = 5;
export type ReconciliationTargetType = "account" | "card";
export type MatchRule = "identity" | "installment" | "exact" | "cents" | "name-period";
export type ReconciliationLine = {
  index: number; identity: { sourceFingerprint: string; externalId: string };
  purchaseDate: string; postedDate: string | null; signedCents: number; name: string;
  installment: { current: number; total: number } | null;
  lineKind: "purchase" | "credit" | "invoice-payment" | null; categoryName: string | null;
};
export type ReconciliationEntry = {
  entryKey: string; // "t:<transactionId>" | "o:<ruleId>:<yyyy-mm-dd>"
  name: string; date: string; period: string; signedCents: number;
  installmentCount: number | null; currentInstallment: number | null;
  externalId: string | null; categoryId: string | null; isSettled: boolean;
  hasSplits: boolean; isTransfer: boolean; isGenerated: boolean; inPaidInvoice: boolean; isRecurring: boolean;
};
export type MatchContext = { targetType: ReconciliationTargetType; nameMappings: Record<string, string> }; // key = getImportedDescriptionKey(line.name)
export type LineClassification = { lineIndex: number } & (
  | { status: "matched"; rule: MatchRule; entryKey: string; divergence: { appCents: number; fileCents: number } | null }
  | { status: "ambiguous"; candidateKeys: string[] }
  | { status: "bank-only" });
export function matchReconciliationLines(input: { lines: ReconciliationLine[]; entries: ReconciliationEntry[]; context: MatchContext }): { lines: LineClassification[]; appOnlyKeys: string[] };
export type MatchSuspicion =
  | { reason: "before-file-start"; entryDate: string; fileStartDate: string }
  | { reason: "amount-divergence"; appCents: number; fileCents: number };
export function detectMatchSuspicions(input: { line: ReconciliationLine; rule: MatchRule; entry: ReconciliationEntry; fileStartDate: string | null; targetType: ReconciliationTargetType }): MatchSuspicion[];
export type AmountAlignment = { kind: "none" } | { kind: "align"; toCents: number }
  | { kind: "kept"; reason: "splits" | "transfer" | "generated" | "paid-invoice" | "card-recurring" };
export function resolveAmountAlignment(input: { line: ReconciliationLine; entry: ReconciliationEntry; targetType: ReconciliationTargetType }): AmountAlignment;
export type ReconciliationBucket = "needs-decision" | "matched" | "bank-only" | "ambiguous" | "ignored" | "informational";
export function assignReconciliationBuckets(input: { lines: ReconciliationLine[]; classification: LineClassification[]; entries: ReconciliationEntry[]; ignoredExternalIds: ReadonlyMap<string, string>; fileStartDate: string | null; targetType: ReconciliationTargetType }): { lineIndex: number; bucket: ReconciliationBucket; suspicions: MatchSuspicion[]; ignoredReason: string | null }[];
export function filterAppOnlyByScope(entries: ReconciliationEntry[], scope: { targetType: "account"; from: string; to: string } | { targetType: "card"; invoicePeriod: string }): ReconciliationEntry[];
export function reclassifyPreviousInvoicePayments(lines: ReconciliationLine[], input: { invoicePeriod: string; dueDay: number }): ReconciliationLine[];
export function matchAccountByIdentification(identification: string | null, accounts: { id: string; name: string }[]): string | null;
export function suggestLineName(line: ReconciliationLine, nameMappings: Record<string, string>): string;

// packages/domain/src/reconciliations-plan.ts
export type LineDecision = { lineIndex: number; action: "keep" | "create" | "link" | "ignore" | "skip";
  entryKey?: string; name?: string; categoryId?: string | null; personId?: string; ignoreReason?: string };
export type EntryRef = { kind: "transaction"; id: string } | { kind: "occurrence"; ruleId: string; date: string };
export type ReconciliationPlan = {
  releases: { ref: EntryRef; externalId: string }[];
  creations: { lineIndex: number; name: string; signedCents: number; purchaseDate: string; period: string; categoryId: string | null; personId: string; externalId: string }[];
  stamps: { ref: EntryRef; externalId: string; lineIndex: number }[];
  alignments: { ref: EntryRef; toCents: number }[];           // entries already settled (account) or card entries
  settlements: { ref: EntryRef; cents: number; date: string }[]; // pending account entries (transfer pairs resolved by the repository)
  ignores: { externalId: string; reason: string }[];
  categoryMappings: { descriptionKey: string; categoryId: string }[]; // unique by key
  nameMappings: { descriptionKey: string; name: string }[];           // unique by key
};
export class ReconciliationPlanError extends Error { code: "RECONCILIATION_STALE" | "RECONCILIATION_NOTHING_TO_APPLY" | "RECONCILIATION_EMPTY_NAME" }
export function buildReconciliationPlan(input: {
  lines: ReconciliationLine[]; classification: LineClassification[]; buckets: ReturnType<typeof assignReconciliationBuckets>;
  entries: ReconciliationEntry[]; decisions: LineDecision[]; targetType: ReconciliationTargetType;
  invoicePeriod: string | null; defaultPersonId: string;
}): ReconciliationPlan; // throws ReconciliationPlanError
export function parseEntryKey(entryKey: string): EntryRef | null;
```
